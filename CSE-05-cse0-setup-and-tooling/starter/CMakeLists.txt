cmake_minimum_required(VERSION 3.20)
project(cse0_starter C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CSE_SANITIZERS "Build this pack with address/undefined diagnostics" OFF)

add_executable(cse0_starter main.c)
add_executable(cse0_starter_tests tests.c main.c)
target_compile_definitions(cse0_starter_tests PRIVATE CSE_NO_MAIN)
foreach(target cse0_starter cse0_starter_tests)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(CSE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    else()
        message(FATAL_ERROR "Use a Clang or GCC C17 toolchain for this pack")
    endif()
endforeach()

# Unfinished learner checks are useful locally but are not a root acceptance gate.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR OR FALSE)
    enable_testing()
    add_test(NAME cse0_starter_contract COMMAND cse0_starter_tests)
endif()
