cmake_minimum_required(VERSION 3.20)
project(cse2_bitflag_parser_solution C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CSE_SANITIZERS "Build with address/undefined diagnostics" OFF)
add_executable(cse2_bitflag_parser_solution main.c input.c configuration.c)
add_executable(cse2_bitflag_parser_solution_tests tests.c input.c configuration.c)
foreach(target cse2_bitflag_parser_solution cse2_bitflag_parser_solution_tests)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
        if(CSE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    else()
        message(FATAL_ERROR "Use a Clang or GCC C17 toolchain")
    endif()
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    enable_testing()
    add_test(NAME cse2_bitflag_parser_solution_contract COMMAND cse2_bitflag_parser_solution_tests)
else()
    add_test(NAME cse2_bitflag_parser_solution_contract COMMAND cse2_bitflag_parser_solution_tests)
endif()
