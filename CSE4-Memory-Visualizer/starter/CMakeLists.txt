cmake_minimum_required(VERSION 3.20)
project(cse4_memory_visualizer_starter C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CSE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(cse4_memory_visualizer_starter_core STATIC layout.c)
target_include_directories(cse4_memory_visualizer_starter_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(cse4_memory_visualizer_starter main.c)
target_link_libraries(cse4_memory_visualizer_starter PRIVATE cse4_memory_visualizer_starter_core)
add_executable(cse4_memory_visualizer_starter_contracts tests.c)
target_link_libraries(cse4_memory_visualizer_starter_contracts PRIVATE cse4_memory_visualizer_starter_core)
foreach(target_name cse4_memory_visualizer_starter_core cse4_memory_visualizer_starter cse4_memory_visualizer_starter_contracts)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(CSE_SANITIZERS)
   target_compile_options(${target_name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target_name} PRIVATE -fsanitize=address,undefined)
  endif()
 elseif(CSE_SANITIZERS)
  message(FATAL_ERROR "This sanitizer profile requires Clang or GCC")
 endif()
endforeach()
include(CTest)
if(BUILD_TESTING AND CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 add_test(NAME cse4_memory_visualizer_starter_contracts COMMAND cse4_memory_visualizer_starter_contracts)
endif()
