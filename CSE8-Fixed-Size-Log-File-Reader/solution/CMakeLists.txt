cmake_minimum_required(VERSION 3.20)
project(cse8_fixed_log_reader_solution C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CSE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(cse8_fixed_log_reader_solution_core STATIC log.c)
target_include_directories(cse8_fixed_log_reader_solution_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(cse8_fixed_log_reader_solution main.c)
target_link_libraries(cse8_fixed_log_reader_solution PRIVATE cse8_fixed_log_reader_solution_core)
add_executable(cse8_fixed_log_reader_solution_contracts tests.c)
target_link_libraries(cse8_fixed_log_reader_solution_contracts PRIVATE cse8_fixed_log_reader_solution_core)
foreach(target_name cse8_fixed_log_reader_solution_core cse8_fixed_log_reader_solution cse8_fixed_log_reader_solution_contracts)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(CSE_SANITIZERS)
   target_compile_options(${target_name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target_name} PRIVATE -fsanitize=address,undefined)
  endif()
 elseif(CSE_SANITIZERS)
  message(FATAL_ERROR "This sanitizer profile requires Clang or GCC")
 endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
 add_test(NAME cse8_fixed_log_reader_solution_contracts COMMAND cse8_fixed_log_reader_solution_contracts)
endif()
