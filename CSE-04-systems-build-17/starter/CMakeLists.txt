cmake_minimum_required(VERSION 3.20)
project(cse17_window_aggregator_starter C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CSE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(cse17_window_aggregator_starter_core STATIC window.c)
target_include_directories(cse17_window_aggregator_starter_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(cse17_window_aggregator_starter main.c)
target_link_libraries(cse17_window_aggregator_starter PRIVATE cse17_window_aggregator_starter_core)
add_executable(cse17_window_aggregator_starter_contracts tests.c)
target_link_libraries(cse17_window_aggregator_starter_contracts PRIVATE cse17_window_aggregator_starter_core)
foreach(target_name cse17_window_aggregator_starter_core cse17_window_aggregator_starter cse17_window_aggregator_starter_contracts)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(CSE_SANITIZERS)
   target_compile_options(${target_name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target_name} PRIVATE -fsanitize=address,undefined)
  endif()
 elseif(CSE_SANITIZERS)
  message(FATAL_ERROR "This sanitizer profile requires Clang or GCC")
 endif()
endforeach()
include(CTest)
if(BUILD_TESTING AND ("starter" STREQUAL "solution" OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR))
 add_test(NAME cse17_window_aggregator_starter_contracts COMMAND cse17_window_aggregator_starter_contracts)
endif()
