cmake_minimum_required(VERSION 3.20)
project(cse15_region_planner_solution C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CSE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(cse15_region_planner_solution_core STATIC regions.c)
target_include_directories(cse15_region_planner_solution_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(cse15_region_planner_solution main.c)
target_link_libraries(cse15_region_planner_solution PRIVATE cse15_region_planner_solution_core)
add_executable(cse15_region_planner_solution_contracts tests.c)
target_link_libraries(cse15_region_planner_solution_contracts PRIVATE cse15_region_planner_solution_core)
foreach(target_name cse15_region_planner_solution_core cse15_region_planner_solution cse15_region_planner_solution_contracts)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(CSE_SANITIZERS)
   target_compile_options(${target_name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target_name} PRIVATE -fsanitize=address,undefined)
  endif()
 elseif(CSE_SANITIZERS)
  message(FATAL_ERROR "This sanitizer profile requires Clang or GCC")
 endif()
endforeach()
include(CTest)
if(BUILD_TESTING AND ("solution" STREQUAL "solution" OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR))
 add_test(NAME cse15_region_planner_solution_contracts COMMAND cse15_region_planner_solution_contracts)
endif()
