cmake_minimum_required(VERSION 3.20)
project(cse1_hex_binary_inspector_starter C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CSE_SANITIZERS "Build this pack with address/undefined diagnostics" OFF)

add_executable(cse1_hex_binary_inspector_starter main.c input.c representation.c)
add_executable(cse1_hex_binary_inspector_starter_tests tests.c input.c representation.c)
foreach(target cse1_hex_binary_inspector_starter cse1_hex_binary_inspector_starter_tests)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
        if(CSE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    else()
        message(FATAL_ERROR "Use a Clang or GCC C17 toolchain for this pack")
    endif()
endforeach()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    enable_testing()
    add_test(NAME cse1_hex_binary_inspector_starter_contract COMMAND cse1_hex_binary_inspector_starter_tests)
endif()
